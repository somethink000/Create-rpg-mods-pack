# Memory: fixed 4 GB. Live after a GC: ~1.1 GB idle, ~1.3 GB generating terrain (mods' nested jars flattened, see
# modpack-docs/performance.md); 3 GB still generated as fast with short pauses, 4 leaves room for players. The process
# takes ~1.3 GB more outside the heap (classes, threads, natives): a machine with 8 GB is enough. Xms = Xmx: no resizing.
-Xms4G
-Xmx4G
# G1 tuned for Minecraft servers (Aikar's flags): short, regular pauses instead of long stalls.
-XX:+UseG1GC
-XX:+ParallelRefProcEnabled
-XX:MaxGCPauseMillis=200
-XX:+UnlockExperimentalVMOptions
-XX:+DisableExplicitGC
-XX:+AlwaysPreTouch
-XX:G1NewSizePercent=30
-XX:G1MaxNewSizePercent=40
-XX:G1HeapRegionSize=8M
-XX:G1ReservePercent=20
-XX:G1HeapWastePercent=5
-XX:SurvivorRatio=32
-XX:+PerfDisableSharedMem
-XX:MaxTenuringThreshold=1
# Profiling (JFR, only while recording): whole call stacks, mods are deep in them.
-XX:FlightRecorderOptions=stackdepth=512
# A 3-core server: background GC marking on one thread, started by G1 itself (not at 15% of the heap,
# which kept it running all the time and took a whole core from the game).
-XX:ConcGCThreads=1
